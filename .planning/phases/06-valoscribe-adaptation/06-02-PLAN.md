         ```python
         def get_current_sides(self, round_number: int) -> dict[str, str]:
             """Get attacker/defender assignment for each team.

             Valorant side logic:
             - Rounds 1-12: First half (starting sides)
             - Rounds 13-24: Second half (swapped)
             - Rounds 25+: Overtime (swap every 2 rounds)
             """
             if round_number <= 12:
                 return dict(self.starting_sides)
             elif round_number <= 24:
                 return self._swap_sides(self.starting_sides)
             else:
                 ot_round = round_number - 24
                 swaps = (ot_round - 1) // 2
                 if swaps % 2 == 0:
                     return self._swap_sides(self.starting_sides)
                 else:
                     return dict(self.starting_sides)
         ```
